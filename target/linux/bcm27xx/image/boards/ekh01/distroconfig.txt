# Keep native Bluetooth on PL011 and header GPS on the mini UART.
# Select Bluetooth-capable BCM2710 boards explicitly: [pi3] also includes
# CM3/CM3+, which have no onboard Bluetooth and keep their PL011 header UART.
# Board types: Pi 3B, 3B+, 3A+, Zero 2 W (Raspberry Pi revision codes).
[board-type=0x8]
dtoverlay=uart1
core_freq=250
[board-type=0xd]
dtoverlay=uart1
core_freq=250
[board-type=0xe]
dtoverlay=uart1
core_freq=250
[board-type=0x12]
dtoverlay=uart1
core_freq=250
[pi3]
dtparam=i2c1=on
[pi4]
dtoverlay=uart1
dtparam=i2c1=on
[cm4]
# This overlay is used to allow m.2 wifi cards
dtoverlay=pcie-32bit-dma
# Enable USB ports on carrier cards
dtoverlay=dwc2,dr_mode=host

[all]
# Disable overscan assuming the display supports displaying the full resolution
# If the text shown on the screen disappears off the edge, comment this out
disable_overscan=1

# How much memory in MB to assign to the GPU on Pi models having
# 256, 512 or 1024 MB total memory
gpu_mem_256=128
gpu_mem_512=128
gpu_mem_1024=128

# enable uart
enable_uart=1

# Do not enable UART5: GPIO12 is the WM1302 GPS wake line.

# disable SD activity LED by default
#(we configure it for HaLow later in userspace)
dtparam=act_led_trigger=none

# adds ramoops device tree overlay
dtoverlay=ramoops

# automatically load sensor dtoverlays
camera_auto_detect=1
