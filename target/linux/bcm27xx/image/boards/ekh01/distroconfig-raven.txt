[all]

#--- Memory / display --------------------------------------------------------
# GPU memory (MB) by total RAM size
gpu_mem_256=128
gpu_mem_512=128
gpu_mem_1024=128
# Display supports full resolution; comment out if edges are cut off
disable_overscan=1
# Headless: blank HDMI output when no display is attached
hdmi_blanking=2

#--- Onboard radios ----------------------------------------------------------
# HaLow radio is the only wireless interface; disable Bluetooth so PL011
# (ttyAMA0, GPIO14/15) is free for the GPS module.
dtoverlay=disable-bt
dtoverlay=disable-wifi

#--- USB / PCIe --------------------------------------------------------------
# Allow PCIe (m.2) wifi cards
dtoverlay=pcie-32bit-dma
# Carrier USB ports run in host mode (MM8108 HaLow radio is on USB)
dtoverlay=dwc2,dr_mode=host

#--- Serial ------------------------------------------------------------------
# Primary UART (PL011 / ttyAMA0, GPIO14/15): LC76G GPS
uart_enable=1
# UART3 (GPIO4/5)  -> ttyAMA1
dtoverlay=uart3
# UART4 (GPIO8/9)  -> ttyAMA2
# Do not add ctsrts: CTS would land on GPIO10 (TALKGROUP_5)
dtoverlay=uart4
# uart5 intentionally NOT enabled: GPIO12 is GPS_WAKE

#--- I2C ---------------------------------------------------------------------
# I2C1 (GPIO2/3) user bus
dtparam=i2c1=on
# I2C0 (GPIO0/1) exposed to userspace (camera / ID EEPROM bus)
dtparam=i2c_vc=on

#--- GPS timing --------------------------------------------------------------
# GPS 1PPS on GPIO18 -> /dev/pps0 (consumed by gpsd/ntp)
dtoverlay=pps-gpio,gpiopin=18

#--- Board overlay -----------------------------------------------------------
# RAVEN carrier: GPIO line names, talk group selector pull-up hog,
# uart5 disabled. Keep after the generic overlays above so it wins any
# pin conflict.
dtoverlay=raven

#--- Misc --------------------------------------------------------------------
# SD activity LED off by default (userspace re-purposes it for HaLow)
dtparam=act_led_trigger=none
# Persist kernel oops/panic logs across reboot
dtoverlay=ramoops
# Auto-load camera sensor overlays
camera_auto_detect=1

# one line at a time.

arm_boost=0
